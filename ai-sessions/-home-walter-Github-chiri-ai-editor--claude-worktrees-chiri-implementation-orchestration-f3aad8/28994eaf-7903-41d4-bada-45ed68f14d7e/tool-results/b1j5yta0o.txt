   160	const readyRanges = (
   161	  suggestion: ReadySuggestion,
   162	  active: boolean,
   163	): readonly Range<Decoration>[] => {
   164	  const { id, from, to, diff } = suggestion
   165	  const hunks = diff.hunks.flatMap((hunk, index): Range<Decoration>[] => {
   166	    if (hunk.type === 'deleted') {
   167	      return hunk.from === hunk.to
   168	        ? []
   169	        : [deletedMark(id).range(from + hunk.from, from + hunk.to)]
   170	    }
   171	    if (hunk.type !== 'inserted') return []
   172	    if (diff.mode === 'block') {
   173	      return [
   174	        Decoration.widget({
   175	          widget: new ReplacementWidget(id, hunk.text),
   176	          block: true,
   177	          side: 1,
   178	        }).range(to),
   179	      ]
   180	    }
   181	    const separated = diff.hunks[index - 1]?.type === 'deleted'
   182	    return [
   183	      Decoration.widget({
   184	        widget: new InsertedWidget(id, hunk.text, separated),
   185	        side: 1,
   186	      }).range(from + hunk.at),
   187	    ]
   188	  })
   189	  return [suggestionMark(id, active).range(from, to), ...hunks]
   190	}

   192	const suggestionRanges = (
   193	  suggestion: Suggestion,
   194	  active: boolean,
   195	): readonly Range<Decoration>[] => {
   196	  switch (suggestion.status) {
   197	    case 'pending':
   198	      return [pendingMark(suggestion.id).range(suggestion.from, suggestion.to)]
   199	    case 'error':
   200	      return [
   201	        failedMark(suggestion.id).range(suggestion.from, suggestion.to),
   202	        Decoration.widget({
   203	          widget: new ErrorCardWidget(
   204	            suggestion.id,
   205	            describeError(suggestion.error),
   206	          ),
   207	          block: true,
   208	          side: 1,
   209	        }).range(suggestion.to),
   210	      ]
   211	    case 'ready':
   212	      return readyRanges(suggestion, active)
   213	  }
   214	}

    43	  create: () => ({ askOpen: false, dismissed: null }),

    32	type Chunk =
    33	  | { readonly type: 'equal'; readonly text: string }
    34	  | {
    35	      readonly type: 'change'
    36	      readonly deleted: string
    37	      readonly inserted: string
    38	    }

    40	const isBlank = (text: string): boolean => text.trim() === ''
    41	
    42	const toChunk = (part: ChangeObject<string>): Chunk =>
    43	  part.added
    44	    ? { type: 'change', deleted: '', inserted: part.value }
    45	    : part.removed
    46	      ? { type: 'change', deleted: part.value, inserted: '' }
    47	      : { type: 'equal', text: part.value }
    48	
    49	const joinChanges = (
    50	  previous: Extract<Chunk, { type: 'change' }>,
    51	  separator: string,
    52	  next: Extract<Chunk, { type: 'change' }>,
    53	): Chunk => ({
    54	  type: 'change',
    55	  deleted:
    56	    previous.deleted + (previous.deleted ? separator : '') + next.deleted,
    57	  inserted:
    58	    previous.inserted + (previous.inserted ? separator : '') + next.inserted,
    59	})
    60	
    61	const pushChunk = (
    62	  chunks: readonly Chunk[],
    63	  chunk: Chunk,
    64	): readonly Chunk[] => {
    65	  if (chunk.type === 'equal') return [...chunks, chunk]
    66	  const last = chunks.at(-1)
    67	  if (last?.type === 'change')
    68	    return [...chunks.slice(0, -1), joinChanges(last, '', chunk)]
    69	  const beforeLast = chunks.at(-2)
    70	  if (
    71	    last?.type === 'equal' &&
    72	    isBlank(last.text) &&
    73	    beforeLast?.type === 'change'
    74	  ) {
    75	    return [...chunks.slice(0, -2), joinChanges(beforeLast, last.text, chunk)]
    76	  }
    77	  return [...chunks, chunk]
    78	}
    79	
    80	const toHunks = (chunks: readonly Chunk[]): readonly DiffHunk[] =>
    81	  chunks.reduce<{ hunks: readonly DiffHunk[]; offset: number }>(
    82	    ({ hunks, offset }, chunk) => {
    83	      if (chunk.type === 'equal') {
    84	        const to = offset + chunk.text.length
    85	        return {
    86	          hunks: [
    87	            ...hunks,
    88	            { type: 'equal', text: chunk.text, from: offset, to },
    89	          ],
    90	          offset: to,
    91	        }
    92	      }
    93	      const to = offset + chunk.deleted.length
    94	      return {
    95	        hunks: [
    96	          ...hunks,
    97	          ...(chunk.deleted === ''
    98	            ? []
    99	            : [
   100	                {
   101	                  type: 'deleted' as const,
   102	                  text: chunk.deleted,
   103	                  from: offset,
   104	                  to,
   105	                },
   106	              ]),
   107	          ...(chunk.inserted === ''
   108	            ? []
   109	            : [{ type: 'inserted' as const, text: chunk.inserted, at: to }]),
   110	        ],
   111	        offset: to,
   112	      }
   113	    },
   114	    { hunks: [], offset: 0 },
   115	  ).hunks
   116	
   117	const countWords = (text: string): number =>
   118	  text.split(/\s+/).filter((word) => word !== '').length
   119	
   120	const sumWords = (hunks: readonly DiffHunk[], type: DiffHunk['type']): number =>
   121	  hunks
   122	    .filter((hunk) => hunk.type === type)
   123	    .reduce((total, hunk) => total + countWords(hunk.text), 0)
   124	
   125	export const changedShare = (hunks: readonly DiffHunk[]): number => {
   126	  const changed = sumWords(hunks, 'deleted') + sumWords(hunks, 'inserted')
   127	  const total = changed + 2 * sumWords(hunks, 'equal')
   128	  return total === 0 ? 0 : changed / total
   129	}
   130	
   131	const blockHunks = (
   132	  original: string,
   133	  replacement: string,
   134	): readonly DiffHunk[] => [
   135	  ...(original === ''
   136	    ? []
   137	    : [
   138	        {
   139	          type: 'deleted' as const,
   140	          text: original,
   141	          from: 0,
   142	          to: original.length,
   143	        },
   144	      ]),
   145	  ...(replacement === ''
   146	    ? []
   147	    : [{ type: 'inserted' as const, text: replacement, at: original.length }]),
   148	]
   149	
   150	export const adaptiveDiff = (
   151	  original: string,
   152	  replacement: string,
   153	): AdaptiveDiff => {
   154	  const hunks = toHunks(
   155	    diffWordsWithSpace(original, replacement)
   156	      .map(toChunk)
   157	      .reduce(pushChunk, []),
   158	  )
   159	  const share = changedShare(hunks)
   160	  return share > BLOCK_LEVEL_THRESHOLD
   161	    ? {
   162	        mode: 'block',
   163	        hunks: blockHunks(original, replacement),
   164	        changedShare: share,
   165	      }
   166	    : { mode: 'word', hunks, changedShare: share }
   167	}
